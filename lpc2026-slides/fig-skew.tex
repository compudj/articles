% P1's fig-skew, re-laid for a 16:9 slide: the same two rows and columns, the
% vertical spacing compressed so it fits under a title with a line above and
% below it, and the deck's palette -- orange for the plain-RCU problem state,
% blue for the entry once reachable.
%
% As in P1, the picture must make one thing obvious: two publications leave a
% MIDDLE state in which a reader finds E by key and finds it in no position,
% and one commit has no middle state at all.
%
% The boxes are the two data structures -- the hash bucket E hashes to, and
% the LRU list -- not nodes embedded in E. They are descriptions, not
% identifiers, so they are set in the sans face.
\begin{tikzpicture}[
  box/.style   = {draw=muted, minimum width=\bw cm, minimum height=5.5mm,
                  fill=paperalt, anchor=west, font=\small},
  ent/.style   = {draw=accent, rounded corners=2pt, minimum width=7mm,
                  minimum height=5.5mm, fill=accent!16},
  gone/.style  = {draw=muted, rounded corners=2pt, minimum width=7mm,
                  minimum height=5.5mm, fill=paper, text=muted, densely dashed},
  arr/.style   = {-{Latex[length=1.8mm]}, thin, ink},
  stub/.style  = {thin, muted},
  hdr/.style   = {font=\small\bfseries, anchor=west},
  note/.style  = {font=\footnotesize, text=muted, align=center},
  skew/.style  = {fill=old!10, draw=old!55, rounded corners=3pt},
  y=0.92cm,
]
% Geometry (cm): column pitch, box width, E's centre from its column's left.
\def\cw{4.6}\def\bw{2.3}\def\ex{3.2}
\pgfmathsetmacro{\cb}{\cw}      % column 2 (between)
\pgfmathsetmacro{\cc}{2*\cw}    % column 3 (after)
\pgfmathsetmacro{\mid}{\cw+\ex/2+0.1}  % centre of the between column

% ===== row 1: two publishes ==================================================
\fill[skew] (\cb-0.25,-0.1) rectangle (\cb+\ex+0.6,1.85);
\node[hdr] at (-0.15,2.2) {Two publishes (plain RCU)};

\node[note] at (\ex/2,1.55) {before};
\node[box] (i1) at (0,1.05) {hash bucket};
\node[box] (r1) at (0,0.3)  {LRU};
\node[gone] (e1) at (\ex,0.675) {$E$};

\node[note, text=old] at (\mid,1.55) {between};
\node[box] (i2) at (\cb,1.05) {hash bucket};
\node[box] (r2) at (\cb,0.3)  {LRU};
\node[ent]  (e2) at (\cb+\ex,0.675) {$E$};
\draw[arr] (i2.east) -- (e2.north west);
\draw[stub] (r2.east) -- ++(0.2,0) node[right, inner sep=1pt, text=old] {$\times$};

\node[note] at (\cc+\ex/2,1.55) {after};
\node[box] (i3) at (\cc,1.05) {hash bucket};
\node[box] (r3) at (\cc,0.3)  {LRU};
\node[ent]  (e3) at (\cc+\ex,0.675) {$E$};
\draw[arr] (i3.east) -- (e3.north west);
\draw[arr] (r3.east) -- (e3.south west);

\node[note, text=old] at (\mid,-0.42)
  {a reader finds $E$ by key, and $E$ is in no position};

% ===== row 2: one commit =====================================================
% Same columns as row 1, so its before/after labels are not repeated.
\node[hdr] at (-0.15,-1.0) {One commit (pseudo-transaction)};

\node[box] (i4) at (0,-1.5)  {hash bucket};
\node[box] (r4) at (0,-2.25) {LRU};
\node[gone] (e4) at (\ex,-1.875) {$E$};

\node[box] (i5) at (\cc,-1.5)  {hash bucket};
\node[box] (r5) at (\cc,-2.25) {LRU};
\node[ent]  (e5) at (\cc+\ex,-1.875) {$E$};
\draw[arr] (i5.east) -- (e5.north west);
\draw[arr] (r5.east) -- (e5.south west);

\draw[-{Latex[length=2.4mm]}, line width=1.2pt, accent]
  (\cb-0.25,-1.875) -- (\cb+\ex+0.6,-1.875)
  node[midway, above, font=\footnotesize] {one commit};
\node[note, text=accent] at (\mid,-2.3) {no middle state};
\end{tikzpicture}
