% Removing node E from P <-> E <-> N via the flip-latch.
% Layout after Desnoyers et al., "User-Level Implementations of Read-Copy
% Update", Fig. 3: time runs left->right (staged | commit | committed),
% viewpoints stack top->bottom (bidirectional reader / updater mechanism),
% and each list draws vertically so the two axes never collide. What that
% figure lacks -- the transaction and its commit -- is exactly the updater
% row here: the two moving edges park on proxies that read ONE selector, and
% the single commit store flips both at once. A reader sees P<->E<->N or
% P<->N, never a mix, in EITHER direction -- the double-headed arrows are one
% coherent structure, not two chains that could disagree.
%
% Self-contained TikZ (no external toolchain), matching fig-fliplatch.tex.
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  font=\small,
  lnode/.style   = {draw, rounded corners=2pt, minimum width=8mm, minimum height=7mm, fill=gray!8},
  ghost/.style   = {draw, rounded corners=2pt, minimum width=8mm, minimum height=7mm, fill=gray!8, dashed, text=gray!55, draw=gray!55},
  slot/.style    = {draw, minimum width=15mm, minimum height=6mm, fill=gray!8},
  proxy/.style   = {draw, minimum width=25mm, minimum height=8mm, fill=blue!6},
  sel/.style     = {draw, very thick, minimum width=16mm, minimum height=8mm, fill=orange!18},
  rowlab/.style  = {align=center, font=\footnotesize\itshape, text=black!75},
  colhead/.style = {align=center, font=\footnotesize\itshape},
  biarr/.style   = {latex-latex, semithick},
  gedge/.style   = {latex-latex, thin, densely dashed, gray!55},
  arr/.style     = {-latex, thin},
  dep/.style     = {-latex, thin, densely dotted, gray!70!black},
]

% ---- column headers --------------------------------------------------------
\node[colhead] at (2.8, 5.35)  {staged\\(\code{sel}$=0$)};
\node[colhead] at (9.4, 5.35)  {committed\\(\code{sel}$=1$)};

% ---- staged | committed boundary: a NEUTRAL time divider, NOT the commit ---
% The commit is the updater's store; it is drawn in the updater row (below),
% acting on the selector, so the reader is never made to look like the actor.
\node[sel] (sel) at (6.8, -0.25) {$0 \to 1$};
\draw[thin, gray!45, densely dashed] (6.8, 5.05) -- (sel.north);

% ---- row separator ---------------------------------------------------------
\draw[thin, gray!45] (-1.6, 1.3) -- (11.2, 1.3);

% ---- row labels ------------------------------------------------------------
\node[rowlab] at (-0.8, 3.4)   {Reader\\(bidirectional)};
\node[rowlab] at (-0.8, -0.45) {Updater\\(mechanism)};

% ===== ROW 1 : the reader's view, both directions at once ===================
% staged (sel = 0): E still on the chain
\node[lnode] (sP) at (2.8, 4.6) {\code{P}};
\node[lnode] (sE) at (2.8, 3.4) {\code{E}};
\node[lnode] (sN) at (2.8, 2.2) {\code{N}};
\draw[biarr] (sP) -- (sE);
\draw[biarr] (sE) -- (sN);
\node[right=1mm of sE, font=\scriptsize, text=black!60] {\code{next}\,/\,\code{prev}};

% committed (sel = 1): E bypassed, its own links left untouched
\node[lnode] (cP) at (9.4, 4.6) {\code{P}};
\node[ghost] (cE) at (9.4, 3.4) {\code{E}};
\node[lnode] (cN) at (9.4, 2.2) {\code{N}};
\draw[gedge] (cP) -- (cE);          % E.prev still -> P
\draw[gedge] (cE) -- (cN);          % E.next still -> N
\draw[biarr, very thick] (cP.east) to[out=0, in=0] (cN.east);
\node[left=1mm of cE, font=\scriptsize, text=gray!60, align=right]
  {\code{E} links\\untouched};

% ===== ROW 2 : the mechanism -- the transaction and its commit ==============
\node[slot]  (t1) at (1.3, 0.3)  {\code{P.next}};
\node[slot]  (t2) at (1.3, -0.8) {\code{N.prev}};

\node[proxy] (p1) at (4.05, 0.3)  {$\{\,\mathit{old}\!:E,\ \mathit{new}\!:N\,\}$};
\node[proxy] (p2) at (4.05, -0.8) {$\{\,\mathit{old}\!:E,\ \mathit{new}\!:P\,\}$};
\draw[arr] (t1) -- (p1);
\draw[arr] (t2) -- (p2);
\draw[dep] (p1.east) -- (sel.north west);
\draw[dep] (p2.east) -- (sel.south west);

% the commit itself: the UPDATER's single store flips the shared selector
\draw[-latex, very thick, orange!70!black] (6.8, -1.55) -- (sel.south);
\node[align=center, font=\footnotesize, text=black] at (6.8, -1.85)
  {\textbf{commit}: one store};

% the resolved effect, on the committed side
\node[right=4mm of sel, align=left, font=\scriptsize] (eff)
  {\code{sel}$=1$:\\[1pt]\code{P.next}\,$\to$\,\code{N}\\\code{N.prev}\,$\to$\,\code{P}};

\end{tikzpicture}
\caption{Removing \code{E} from \code{P}\,$\leftrightarrow$\,\code{E}\,$\leftrightarrow$\,\code{N}
through the flip-latch, laid out as in \citet{desnoyers2012urcu}: time runs
left to right, the reader's view sits above the updater's mechanism, and the
single commit store is the vertical line between the two states. The two
reader-visible edges that move --- the forward \code{P.next} ($E\!\to\!N$) and
the backward \code{N.prev} ($E\!\to\!P$) --- park on two proxies that read one
shared selector (updater row). While \code{sel} is $0$ both resolve to \code{E},
so every reader still traverses
\code{P}\,$\leftrightarrow$\,\code{E}\,$\leftrightarrow$\,\code{N}. The commit is
a single release store, $0\!\to\!1$, after which both resolve to their new
targets and every reader traverses \code{P}\,$\leftrightarrow$\,\code{N}. The
double-headed arrows are one coherent structure, not two independent chains:
there is no instant, and no direction, at which the forward and backward views
disagree. Only these two neighbour edges join the transaction; \code{E} itself
does not --- its own \code{next} and \code{prev} are left pointing at \code{N}
and \code{P} (dashed), so a reader standing on \code{E} when the commit lands
reads straight through them and continues in either direction.}
\label{fig:listremove}
\end{figure}
